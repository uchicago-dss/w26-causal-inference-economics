\begin{frame}{Background}
  \small
  \begin{itemize}
    \item China's economy grew rapidly in the 1990s, becoming a major exporter to the U.S.
    \item Under U.S.\ law, China received low tariff rates (Normal Trade Relations) but Congress had to \textbf{vote to renew} NTR status every year.
    \item This annual renewal created \textbf{trade-policy uncertainty}
          \begin{itemize}
            \item importers faced the risk that tariffs could spike to 1930s-era levels if Congress voted no.
          \end{itemize}
    \item In \textbf{October 2000}, the U.S.\ passed the law granting China Permanent Normal Trade Relations (PNTR), conditional on WTO accession.
    \item In \textbf{December 2001}, China officially joined the WTO; permanent low tariffs took effect in January 2002.
    \item We take advantage of the fact that the actual tariff rates mostly \textbf{did not change}: what changed was the \textbf{elimination of uncertainty} about future tariffs.
  \end{itemize}
\end{frame}

\begin{frame}{Policy Timing and Identification Intuition}
  \small
  \begin{itemize}
    \item \textbf{Pre-2001}: China receives NTR annually; risk of non-renewal creates uncertainty that is priced into trade decisions.
    \item \textbf{2000}: U.S.\ grants PNTR (legal permanence, conditional on WTO accession).
    \item \textbf{Dec 2001 / Jan 2002}: China joins WTO; permanent low tariffs take effect.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Why did uncertainty matter differently across products?}
  \begin{itemize}
    \item Every product had two legislated tariff rates: a low NTR rate and a high non-NTR rate (dating back to the Smoot-Hawley Act of 1930).
    \item The \textbf{NTR gap} = non-NTR rate $-$ NTR rate measures how much a product's tariff \emph{would have jumped} if Congress didn't renew.
    \item Products with large gaps faced more risk $\rightarrow$ importers could be heavily impacted.
    \item Products with small gaps faced less risk $\rightarrow$ less impact by the annual vote.
  \end{itemize}
\end{frame}

\begin{frame}{Identification Strategy: Difference-in-Differences}
  \small
  \begin{block}{Core idea}
    Compare import growth for \textbf{high-gap} products (more exposed to uncertainty) vs.\ \textbf{low-gap} products (less exposed), before and after WTO entry.
  \end{block}

  \vspace{0.2cm}
  \textbf{Why this works:}
  \begin{itemize}
    \item Low-gap products serve as a \textbf{control group}: because, while they were still affected, they had a small stake from the policy change.
    \item Any ``extra'' import growth in high-gap products after 2002, beyond what low-gap products experienced, can be attributed to the removal of tariff uncertainty.
  \end{itemize}


  % edit this slide to more explicit about the treatment and control groups and that was assume that only difference between the treatment and control group (after our controls) is the policy change
  \vspace{0.2cm}
  \textbf{Treatment exposure (measured in 1999, before WTO negotiations):}
  \begin{equation*}
    \text{Exposure}_p = \text{gap\_pre}_p = (\text{non-NTR tariff})_p - (\text{NTR tariff})_p
  \end{equation*}

  Products above the median gap $\rightarrow$ ``high-gap'' (treatment); below $\rightarrow$ ``low-gap'' (control).
\end{frame}

\begin{frame}{Takeaways and Next Steps}
  \small
  \textbf{Takeaways}
  \begin{itemize}
    \item A $10\%$ increase in exposure causes an estimated $100(\exp(\hat\beta\cdot 0.10)-1)\approx 1.3\%$ post-policy decline in "price" (unit value).
          \begin{itemize}
            \item At the median exposure ($gap\_pre=0.311$), the implied post-policy decline is $100(\exp(\hat\beta\cdot 0.311)-1)\approx 4.0\%$.
            \item Interpreted as the effect of removing trade-policy uncertainty on prices
          \end{itemize}
  \end{itemize}

  \textbf{Next steps}
  \begin{itemize}
    \item Further finetune this parameter by:
          \begin{itemize}
            \item seasonality controls
            \item solve heterogeneity of products issue
            \item controls for other events that occured
            \item compare how imports changed for another country in their trade with China
          \end{itemize}
  \end{itemize}
\end{frame}
